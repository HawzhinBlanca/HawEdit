"""pytest plugin (research only): log every real face-probe result with the test that caused it.

Changes nothing: it calls the real probe and appends one line per call to $WHATIF_LOG. A line with
face_detected_share > 0 and median_face_height_share < 0.05 is a measurement the proposed floor
would change.
"""

import os
from pathlib import Path

import hawedit.measure as m

_orig = m.probe_face_tracking
_log = Path(os.environ["WHATIF_LOG"])


def _logged(video_path, *args, **kwargs):  # type: ignore[no-untyped-def]
    result = _orig(video_path, *args, **kwargs)
    test = os.environ.get("PYTEST_CURRENT_TEST", "?")
    with _log.open("a", encoding="utf-8") as fh:
        fh.write(
            f"{test}\tshare={result.face_detected_share}\tmedian_h={result.median_face_height_share}"
            f"\tfirst={result.first_frame_face_share}\tspeaking={result.speaking_face_share}"
            f"\tsamples={result.samples_count}\n"
        )
    return result


m.probe_face_tracking = _logged
